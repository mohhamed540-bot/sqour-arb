const { DEFAULT_ROOM } = require('./config'); const { verifyToken } = require('./auth'); const { findUserById, getRoomById, getRoomMessages, addMessage, isUserBanned, ensureRoomMember, getMessageReactions, toggleMessageReaction } = require('./db'); function sanitizeText(value, fallback = '') { if (typeof value !== 'string') return fallback; const text = value.trim(); return text || fallback; } function registerSocket(io, socket) { socket.on('auth', (payload = {}) => { const token = payload.token || payload.authorization || ''; const decoded = verifyToken(token); if (!decoded) { socket.emit('auth_error', { message: 'Invalid token' }); socket.disconnect(true); return; } const user = findUserById(decoded.sub); if (!user) { socket.emit('auth_error', { message: 'User not found' }); socket.disconnect(true); return; } if (isUserBanned(user.id)) { socket.emit('auth_error', { message: 'This user is banned' }); socket.disconnect(true); return; } socket.user = user; socket.join(DEFAULT_ROOM); ensureRoomMember(DEFAULT_ROOM, user.id); socket.emit('auth_ok', { user: { id: user.id, username: user.username, displayName: user.displayName, role: user.role, status: user.status, avatar: user.avatar, themeColor: user.themeColor, bio: user.bio }, roomId: DEFAULT_ROOM }); io.emit('presence', { type: 'online', user: { id: user.id, username: user.username, role: user.role }, roomId: DEFAULT_ROOM }); }); socket.on('join-room', (roomId) => { if (!socket.user) { socket.emit('error', { message: 'Please authenticate first' }); return; } const room = getRoomById(roomId || DEFAULT_ROOM); if (!room) { socket.emit('error', { message: 'Room not found' }); return; } socket.join(room.id); ensureRoomMember(room.id, socket.user.id); socket.emit('room-history', { roomId: room.id, messages: getRoomMessages(room.id).map((message) => ({ ...message, reactions: getMessageReactions(message.id) })) }); io.to(room.id).emit('presence', { type: 'join', roomId: room.id, user: { id: socket.user.id, username: socket.user.username, displayName: socket.user.displayName, role: socket.user.role } }); }); socket.on('send-message', ({ roomId, text, replyTo } = {}) => { if (!socket.user) { socket.emit('error', { message: 'Please authenticate first' }); return; } if (isUserBanned(socket.user.id)) { socket.emit('error', { message: 'This user is banned and cannot send messages' }); return; } const room = getRoomById(roomId || DEFAULT_ROOM); if (!room) { socket.emit('error', { message: 'Room not found' }); return; } const cleanText = sanitizeText(text, ''); if (!cleanText) { socket.emit('error', { message: 'Message text is required' }); return; } const message = addMessage({ roomId: room.id, userId: socket.user.id, username: socket.user.username, text: cleanText, replyTo: replyTo || null }); const payload = { ...message, reactions: [], user: { id: socket.user.id, username: socket.user.username, displayName: socket.user.displayName, role: socket.user.role, avatar: socket.user.avatar, themeColor: socket.user.themeColor } }; io.to(room.id).emit('new-message', payload); }); socket.on('reaction', ({ roomId, messageId, emoji } = {}) => { if (!socket.user) return; const room = getRoomById(roomId || DEFAULT_ROOM); if (!room) return; const result = toggleMessageReaction({ messageId, userId: socket.user.id, emoji }); const reactions = getMessageReactions(messageId); io.to(room.id).emit('message-reacted', { roomId: room.id, messageId, reactions, action: result.action, emoji: result.emoji, userId: socket.user.id }); }); socket.on('typing', ({ roomId, isTyping = true } = {}) => { if (!socket.user) return; const room = getRoomById(roomId || DEFAULT_ROOM); if (!room) return; socket.to(room.id).emit('typing', { roomId: room.id, userId: socket.user.id, username: socket.user.username, isTyping }); }); socket.on('disconnect', () => { if (socket.user) { io.emit('presence', { type: 'offline', user: { id: socket.user.id, username: socket.user.username } }); } }); } module.exports = { registerSocket };
